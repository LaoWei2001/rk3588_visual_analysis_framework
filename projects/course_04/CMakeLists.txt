cmake_minimum_required(VERSION 3.12)
# Select a shared engine; external projects can pass -DRKVISION_ENGINE=/path/to/framework.
if(NOT RKVISION_ENGINE)
    if(DEFINED ENV{RKVISION_ENGINE} AND NOT "$ENV{RKVISION_ENGINE}" STREQUAL "")
        set(RKVISION_ENGINE "$ENV{RKVISION_ENGINE}")
    else()
        include("${CMAKE_CURRENT_LIST_DIR}/engine.local.cmake" OPTIONAL)
    endif()
endif()
if(NOT RKVISION_ENGINE)
    get_filename_component(RKVISION_ENGINE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
endif()
set(RKVISION_ENGINE "${RKVISION_ENGINE}" CACHE PATH "RKVision engine checkout")
if(NOT EXISTS "${RKVISION_ENGINE}/vision_analysis/CMakeLists.txt")
    message(FATAL_ERROR "Set RKVISION_ENGINE to your framework checkout")
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
project(visual_application LANGUAGES C CXX)
set(VISION_PROJECT_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
add_subdirectory("${RKVISION_ENGINE}/vision_analysis" "${CMAKE_CURRENT_BINARY_DIR}/engine")
