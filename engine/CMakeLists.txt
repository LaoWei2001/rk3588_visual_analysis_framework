cmake_minimum_required(VERSION 3.10.2)

# Build with: cmake -S engine -B build/engine
# Cross compilation must select the toolchain before project().
include(cmake/Toolchain.cmake)
project(vision_analysis LANGUAGES C CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-Wall -Wpointer-arith -O3)

option(VISION_INFERENCE_ONLY "Build inference tools without a business workspace or video application" OFF)
option(VISION_BUILD_APPLICATION "Build the configured business application and its tests" ON)
include(cmake/Dependencies.cmake)
include(cmake/SDK.cmake)
include(cmake/Network.cmake)
include(cmake/ModelInspection.cmake)
include(cmake/Inference.cmake)
if(NOT VISION_INFERENCE_ONLY)
    include(cmake/Analysis.cmake)
    include(cmake/Engine.cmake)
    if(VISION_BUILD_APPLICATION)
        include(cmake/Business.cmake)
        include(cmake/Application.cmake)
        include(cmake/Tests.cmake)
    endif()
endif()
